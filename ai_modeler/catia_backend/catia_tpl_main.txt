def main():
    try:
        import win32com.client as win32
    except ImportError:
        print('缺少 pywin32, 请先执行:  pip install pywin32')
        return 1

    print('正在连接 CATIA ...')
    try:
        catia = win32.Dispatch('CATIA.Application')
    except Exception as e:
        print('无法启动 CATIA: %s' % e)
        print('请确认本机已安装 CATIA V5。')
        return 1
    try:
        catia.Visible = True
        catia.DisplayFileAlerts = False
    except Exception:
        pass

    doc = catia.Documents.Add('Part')
    part = doc.Part
    part.Name = PART_NAME

    ok, failed = 0, []
    for i, entry in enumerate(FEATURES, 1):
        label = entry.get('name') or entry['kind']
        print('  [%d/%d] %s ...' % (i, len(FEATURES), label))
        try:
            build_body(part, entry)
            part.Update()
            ok += 1
        except Exception as e:
            failed.append('%s -> %s' % (label, e))
            print('       失败: %s' % e)
    try:
        part.Update()
    except Exception:
        pass

    try:
        doc.SaveAs(OUTPUT_CATPart)
        print('已保存: ' + OUTPUT_CATPart)
    except Exception as e:
        print('保存失败 (可在 CATIA 中手动另存): %s' % e)

    print('-' * 60)
    print('完成 %d / %d 个特征' % (ok, len(FEATURES)))
    if failed:
        print('以下特征需要手动处理:')
        for item in failed:
            print('   - ' + item)
    return 0 if not failed else 2


if __name__ == '__main__':
    sys.exit(main())
